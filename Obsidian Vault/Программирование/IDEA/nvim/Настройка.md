# Полный гайд по настройке Neovim для C++ разработки

---

## 1. Установка Neovim

```bash
# Ubuntu/Debian
sudo apt update
sudo apt install neovim

# Arch
sudo pacman -S neovim

# Проверить версию (нужна >= 0.11)
nvim --version
```

---

## 2. Структура конфигурации

Все файлы конфигурации лежат в `~/.config/nvim/`:

```
~/.config/nvim/
├── init.lua                 # Главный файл (загрузка lazy.nvim)
└── lua/
    └── plugins/
        ├── lsp.lua          # LSP (clangd)
        ├── tools.lua        # Автодополнение, подсветка
        └── git.lua          # Git-интеграция
        └── theme.lua        # Тема для редактора
```

Создай эту структуру:

```bash
mkdir -p ~/.config/nvim/lua/plugins
touch ~/.config/nvim/init.lua
touch ~/.config/nvim/lua/plugins/lsp.lua
touch ~/.config/nvim/lua/plugins/tools.lua
touch ~/.config/nvim/lua/plugins/git.lua
touch ~/.config/nvim/lua/plugins/theme.lua
```

## 3. init.lua — загрузка lazy.nvim

```lua
-- ~/.config/nvim/init.lua

-- Установка lazy.nvim (менеджер плагинов)
local lazypath = vim.fn.stdpath("data") .. "/lazy/lazy.nvim"
if not (vim.uv or vim.loop).fs_stat(lazypath) then
  local lazyrepo = "https://github.com/folke/lazy.nvim.git"
  local out = vim.fn.system({ "git", "clone", "--filter=blob:none", "--branch=stable", lazyrepo, lazypath })
  if vim.v.shell_error ~= 0 then
    vim.api.nvim_echo({
      { "Failed to clone lazy.nvim:\n", "ErrorMsg" },
      { out, "WarningMsg" },
      { "\nPress any key to exit..." },
    }, true, {})
    vim.fn.getchar()
    os.exit(1)
  end
end
vim.opt.rtp:prepend(lazypath)

-- Настройка leader клавиши (пробел)
vim.g.mapleader = " "

-- Подключение плагинов из папки lua/plugins
require("lazy").setup({
  spec = {
    { import = "plugins" }, -- Автоматически подгружает все файлы из lua/plugins
  },
})

-- ========== НАСТРОЙКА ВИЗУАЛЬНОЙ ПОДСВЕТКИ ОШИБОК ==========

-- Включаем подсветку ошибок
vim.diagnostic.config({
  virtual_text = true,        -- Показывать текст ошибки рядом со строкой
  signs = true,               -- Показывать иконки на полях (слева)
  underline = true,           -- Подчеркивать ошибки волнистой линией
  update_in_insert = true,    -- Обновлять при вводе текста
  severity_sort = true,       -- Сортировать ошибки по важности
  float = {
    border = "rounded",       -- Красивые границы у окна с ошибкой
    source = true,            -- Показывать источник ошибки (clang)
  },
})

-- Настройка иконок для ошибок (на левой панели)
local signs = {
  { name = "DiagnosticSignError", text = "✘" },
  { name = "DiagnosticSignWarn",  text = "▲" },
  { name = "DiagnosticSignHint",  text = "⚑" },
  { name = "DiagnosticSignInfo",  text = "" },
}

for _, sign in ipairs(signs) do
  vim.fn.sign_define(sign.name, { texthl = sign.name, text = sign.text, numhl = "" })
end

-- Клавиши для навигации по ошибкам (дополнительно к уже существующим)
vim.keymap.set('n', '<leader>e', vim.diagnostic.open_float, { desc = "Показать ошибку под курсором" })

-- ========== НАСТРОЙКИ ИНТЕРФЕЙСА ==========

-- Включаем нумерацию строк
vim.opt.number = true        -- Абсолютные номера строк
   
```

## lsp.lua

```lua
return {
  -- Mason: установка LSP серверов
  {
    "williamboman/mason.nvim",
    build = ":MasonUpdate",
    config = function()
      require("mason").setup()
    end,
  },
  -- Подключение Mason к LSPConfig
  {
    "williamboman/mason-lspconfig.nvim",
    dependencies = { "williamboman/mason.nvim" },
    config = function()
      require("mason-lspconfig").setup({
        ensure_installed = { "clangd" },
        automatic_installation = true,
      })
    end,
  },
  -- Новая настройка LSP через vim.lsp.config
  {
    "neovim/nvim-lspconfig",
    dependencies = { "williamboman/mason-lspconfig.nvim" },
    config = function()
      -- НАСТРАИВАЕМ КЛАНГД
      vim.lsp.config("clangd", {
        cmd = { 
                "clangd",
        },
        filetypes = { "c", "cpp", "objc", "objcpp" },
        root_markers = { "compile_commands.json" },
      })

      -- ПОДКЛЮЧАЕМ КЛИЕНТ К БУФЕРАМ
      vim.api.nvim_create_autocmd("LspAttach", {
        group = vim.api.nvim_create_augroup("UserLspConfig", {}),
        callback = function(ev)
          -- Горячие клавиши
          local opts = { buffer = ev.buf }
          vim.keymap.set("n", "gd", vim.lsp.buf.definition, opts)
          vim.keymap.set("n", "K", vim.lsp.buf.hover, opts)
          vim.keymap.set("n", "gr", vim.lsp.buf.references, opts)
          vim.keymap.set("n", "<leader>rn", vim.lsp.buf.rename, opts)
          vim.keymap.set("n", "<leader>ca", vim.lsp.buf.code_action, opts)
          vim.keymap.set("n", "[d", vim.diagnostic.goto_prev, opts)
          vim.keymap.set("n", "]d", vim.diagnostic.goto_next, opts)
        end,
      })

      -- ЗАПУСКАЕМ ВСЕ LSP КОНФИГИ
      vim.lsp.enable("clangd")
    end,
  },
}
```

## 5. tools.lua — автодополнение и подсветка

```lua
-- ~/.config/nvim/lua/plugins/tools.lua
return {
  -- ========== ПОДСВЕТКА СИНТАКСИСА (TREESITTER) ==========
  {
    "nvim-treesitter/nvim-treesitter",
    build = ":TSUpdate",
    config = function()
      local ok, treesitter = pcall(require, "nvim-treesitter.configs")
      if ok then
        treesitter.setup({
          ensure_installed = { "c", "cpp", "cmake", "make", "bash", "lua", "vim" },
          auto_install = true,
          highlight = {
            enable = true,
          },
          indent = {
            enable = true,
          },
        })
      end
    end,
  },

  -- ========== АВТОДОПОЛНЕНИЕ ==========
  {
    "hrsh7th/nvim-cmp",
    dependencies = {
      "hrsh7th/cmp-nvim-lsp",
      "hrsh7th/cmp-buffer",
      "hrsh7th/cmp-path",
      "hrsh7th/cmp-cmdline",
      "L3MON4D3/LuaSnip",
    },
    config = function()
      local cmp = require("cmp")
      local luasnip = require("luasnip")

      cmp.setup({
        snippet = {
          expand = function(args)
            luasnip.lsp_expand(args.body)
          end,
        },
        mapping = cmp.mapping.preset.insert({
          ["<Tab>"] = cmp.mapping.select_next_item(),
          ["<S-Tab>"] = cmp.mapping.select_prev_item(),
          ["<CR>"] = cmp.mapping.confirm({ select = true }),
        }),
        sources = cmp.config.sources({
          { name = "nvim_lsp" },
          { name = "buffer" },
          { name = "path" },
        }),
      })

      cmp.setup.cmdline(":", {
        sources = {
          { name = "cmdline" },
        },
      })
    end,
  },
}
```

##  git.lua — Git-интеграция

```lua
-- ~/.config/nvim/lua/plugins/git.lua
return {
  -- ========== ПОКАЗЫВАЕТ ИЗМЕНЕНИЯ В ФАЙЛЕ (gitsigns) ==========
  {
    "lewis6991/gitsigns.nvim",
    config = function()
      require("gitsigns").setup({
        signs = {
          add = { text = "│" },
          change = { text = "│" },
          delete = { text = "_" },
          topdelete = { text = "‾" },
          changedelete = { text = "~" },
          untracked = { text = "│" },
        },
        on_attach = function(bufnr)
          local gs = package.loaded.gitsigns

          -- Горячие клавиши для gitsigns
          vim.keymap.set("n", "]c", function()
            if vim.wo.diff then
              vim.cmd.normal({ "]c", bang = true })
            else
              gs.nav_hunk("next")
            end
          end, { buffer = bufnr, desc = "Перейти к следующему изменению" })

          vim.keymap.set("n", "[c", function()
            if vim.wo.diff then
              vim.cmd.normal({ "[c", bang = true })
            else
              gs.nav_hunk("prev")
            end
          end, { buffer = bufnr, desc = "Перейти к предыдущему изменению" })

          -- Действия над изменениями
          vim.keymap.set("n", "<leader>hs", gs.stage_hunk, { buffer = bufnr, desc = "Добавить изменение в stage" })
          vim.keymap.set("n", "<leader>hr", gs.reset_hunk, { buffer = bufnr, desc = "Отменить изменение" })
          vim.keymap.set("n", "<leader>hS", gs.stage_buffer, { buffer = bufnr, desc = "Добавить весь файл в stage" })
          vim.keymap.set("n", "<leader>hu", gs.undo_stage_hunk, { buffer = bufnr, desc = "Убрать из stage" })
          vim.keymap.set("n", "<leader>hp", gs.preview_hunk, { buffer = bufnr, desc = "Показать изменения" })
          vim.keymap.set("n", "<leader>hb", function()
            gs.blame_line({ full = true })
          end, { buffer = bufnr, desc = "Кто написал эту строку" })
          vim.keymap.set("n", "<leader>hd", gs.diffthis, { buffer = bufnr, desc = "Показать diff" })
          vim.keymap.set("n", "<leader>hD", function()
            gs.diffthis("~")
          end, { buffer = bufnr, desc = "Показать diff с предыдущим коммитом" })
        end,
      })
    end,
  },

  -- ========== GIT КЛИЕНТ ВНУТРИ NEOVIM (neogit) ==========
  {
    "NeogitOrg/neogit",
    dependencies = {
      "nvim-lua/plenary.nvim", -- Нужен для работы neogit
      "sindrets/diffview.nvim", -- Показывает diff в удобном интерфейсе
    },
    config = function()
      local neogit = require("neogit")
      neogit.setup({
        -- Настройки интерфейса
        kind = "split", -- Открывать в сплите (можно "vsplit", "tab")
        disable_commit_confirmation = false,
        disable_push_confirmation = false,
        -- Визуальные настройки
        signs = {
          section = { ">", "v" },
          item = { ">", "v" },
          hunk = { "", "" },
        },
        integrations = {
          diffview = true, -- Использовать diffview для показа diff
        },
        -- Автоматически обновлять статус
        auto_refresh = true,
      })

      -- Горячая клавиша для открытия neogit
      vim.keymap.set("n", "<leader>gg", ":Neogit<CR>", { desc = "Открыть Git клиент" })
      vim.keymap.set("n", "<leader>gG", ":Neogit cwd=%:p:h<CR>", { desc = "Открыть Git клиент в папке файла" })
    end,
  },

  -- ========== PLAINARY (зависимость для neogit) ==========
  {
    "nvim-lua/plenary.nvim",
  },

  -- ========== DIFFVIEW (показывает изменения красиво) ==========
  {
    "sindrets/diffview.nvim",
    config = function()
      vim.keymap.set("n", "<leader>gd", ":DiffviewOpen<CR>", { desc = "Открыть DiffView" })
      vim.keymap.set("n", "<leader>gD", ":DiffviewClose<CR>", { desc = "Закрыть DiffView" })
    end,
  },
}
```

## Сводная таблица горячих клавиш

| Клавиша      | Что делает                                      |
| ------------ | ----------------------------------------------- |
| `gd`         | Перейти к определению                           |
| `K`          | Показать описание под курсором                  |
| `gr`         | Найти все ссылки на символ                      |
| `<leader>rn` | Переименовать символ                            |
| `<leader>ca` | Рефакторинг (Code Action)                       |
| `[d`         | Перейти к предыдущей ошибке                     |
| `]d`         | Перейти к следующей ошибке                      |
| `<leader>e`  | Показать ошибку под курсором (всплывающее окно) |
| `<leader>f`  | Форматировать код (clang-format)                |
| `<leader>hs` | Добавить изменение в Git stage                  |
| `<leader>hr` | Отменить изменение                              |
| `<leader>hb` | Показать blame (кто написал строку)             |
| `<leader>gg` | Открыть Git клиент (neogit)                     |
| `<leader>gd` | Открыть DiffView                                |
| `]c`         | Перейти к следующему изменению в файле          |
| `[c`         | Перейти к предыдущему изменению в файле         |
